#
# Per-mapping workflow transitions (one row = one allowed transition).
#
# Format:  ROLE, CURRENT_STATUS, ACTION, RESULT_STATUS
#
#   ROLE            Map project role: SPECIALIST, LEAD, or ADMIN
#   CURRENT_STATUS  MapWorkflowStatus phase before the action (who owns it is in assigned_user, not the status name)
#   ACTION          MappingWorkflowAction taken by that role
#   RESULT_STATUS   MapWorkflowStatus phase after a successful transition
#
# Phase glossary (simple path):
#   NEW                  In scope; unassigned; not edited this mapset version
#   EDITING_IN_PROGRESS  Assigned to a specialist; actively being edited
#   EDITING_DONE         Specialist submitted; awaiting lead approval
#   READY_FOR_PUBLICATION  Mapping accepted; no further mapping work this cycle
#
# Loaded at startup by MappingWorkflowService.resolveTransition(role, status, action).
# No matching row => transition is not allowed (returns null).
#

##################
# SPECIALIST — assign, release, and finish editing
##################

# Specialist self-assigns an unassigned mapping from the queue
SPECIALIST,NEW,ASSIGN,EDITING_IN_PROGRESS

# Specialist gives up the assignment without submitting (clears assigned_user)
SPECIALIST,EDITING_IN_PROGRESS,RELEASE,NEW

# Specialist submits completed mapping (clears assigned_user; may trigger review/conflict routing later)
SPECIALIST,EDITING_IN_PROGRESS,FINISH_EDITING,EDITING_DONE

##################
# LEAD — approve finished work for publication readiness
##################

# Lead accepts specialist work; mapping is ready when the mapset publishes
LEAD,EDITING_DONE,APPROVE_FOR_PUBLICATION,READY_FOR_PUBLICATION

##################
# ADMIN — override assignment without changing phase (except force-release)
##################

# Admin clears assignment and returns concept to the queue
ADMIN,EDITING_IN_PROGRESS,FORCE_RELEASE,NEW

# Admin assigns the mapping to another user; phase stays in progress
ADMIN,EDITING_IN_PROGRESS,REASSIGN,EDITING_IN_PROGRESS

# Lead reassigns the mapping to another specialist
LEAD,EDITING_IN_PROGRESS,REASSIGN,EDITING_IN_PROGRESS

##################
# REVIEW PROJECT — lead review after specialist finish
##################

# Lead picks up a mapping awaiting review
LEAD,REVIEW_NEEDED,START_REVIEW,REVIEW_IN_PROGRESS

# Lead accepts review
LEAD,REVIEW_IN_PROGRESS,ACCEPT_REVIEW,REVIEW_RESOLVED

# Lead rejects review; mapping returns to the queue (assignment cleared)
LEAD,REVIEW_IN_PROGRESS,REJECT_REVIEW,NEW

# Lead sends mapping back for specialist revision
LEAD,REVIEW_IN_PROGRESS,REQUEST_REVISION,EDITING_IN_PROGRESS

# Lead approves reviewed mapping for publication readiness
LEAD,REVIEW_RESOLVED,APPROVE_FOR_PUBLICATION,READY_FOR_PUBLICATION

##################
# CONFLICT PROJECT — lead conflict resolution after both slots finish
##################

# Lead picks up a mapping with detected conflict
LEAD,CONFLICT_DETECTED,START_CONFLICT_RESOLUTION,CONFLICT_IN_PROGRESS

# Lead resolves conflict; mapping is ready for publication
LEAD,CONFLICT_IN_PROGRESS,RESOLVE_CONFLICT,READY_FOR_PUBLICATION
